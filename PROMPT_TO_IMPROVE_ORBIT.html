    /** Generates high-fidelity procedural hardcover canvas textures (Front, Spine, Back) */
    function generateCoverTexture(item) {
      const W = 1024;
      const H = 1480;
      const canvas = document.createElement('canvas');
      canvas.width = W;
      canvas.height = H;
      const ctx = canvas.getContext('2d');

      // Base cloth / matte paper tone
      ctx.fillStyle = item.caseColor;
      ctx.fillRect(0, 0, W, H);

      // Micro cloth grain simulation
      const imgData = ctx.getImageData(0, 0, W, H);
      const d = imgData.data;
      for (let i = 0; i < d.length; i += 4) {
        const noise = (Math.random() - 0.5) * 9;
        d[i] = Math.min(255, Math.max(0, d[i] + noise));
        d[i+1] = Math.min(255, Math.max(0, d[i+1] + noise));
        d[i+2] = Math.min(255, Math.max(0, d[i+2] + noise));
      }
      ctx.putImageData(imgData, 0, 0);

      // Subtle vignette / lighting falloff
      const grad = ctx.createRadialGradient(W * 0.5, H * 0.45, 120, W * 0.5, H * 0.5, W * 0.75);
      grad.addColorStop(0, 'rgba(255,255,255,0.06)');
      grad.addColorStop(1, 'rgba(0,0,0,0.08)');
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, W, H);

      // Editorial typography
      ctx.fillStyle = item.ink;
      ctx.textAlign = 'center';
      
      // Series Mark
      ctx.font = '600 22px system-ui, -apple-system, sans-serif';
      ctx.letterSpacing = '0.12em';
      ctx.fillText('BELIEF CHANGER', W / 2, 180);

      // Minimalist specimen drawing placeholder
      ctx.save();
      ctx.strokeStyle = item.ink;
      ctx.lineWidth = 3.5;
      ctx.globalAlpha = 0.75;
      ctx.beginPath();
      ctx.arc(W / 2, H * 0.46, 90, 0, Math.PI * 2);
      ctx.stroke();
      ctx.beginPath();
      ctx.arc(W / 2, H * 0.46, 6, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();

      // Title
      ctx.font = '700 52px system-ui, -apple-system, sans-serif';
      ctx.letterSpacing = '-0.02em';
      ctx.fillText(item.title, W / 2, H * 0.74);

      // Subtitle
      ctx.font = '400 24px Georgia, serif';
      ctx.fontStyle = 'italic';
      ctx.globalAlpha = 0.7;
      ctx.fillText('A short book of clarity', W / 2, H * 0.79);

      const texture = new THREE.CanvasTexture(canvas);
      texture.anisotropy = 8;
      return texture;
    }

    /** Procedural book spine texture */
    function generateSpineTexture(item) {
      const W = 256;
      const H = 1480;
      const canvas = document.createElement('canvas');
      canvas.width = W;
      canvas.height = H;
      const ctx = canvas.getContext('2d');

      ctx.fillStyle = item.caseColor;
      ctx.fillRect(0, 0, W, H);

      // Spine edge shadow grooves
      const shadowGrad = ctx.createLinearGradient(0, 0, W, 0);
      shadowGrad.addColorStop(0, 'rgba(0,0,0,0.22)');
      shadowGrad.addColorStop(0.18, 'rgba(0,0,0,0.03)');
      shadowGrad.addColorStop(0.5, 'rgba(255,255,255,0.08)');
      shadowGrad.addColorStop(0.82, 'rgba(0,0,0,0.03)');
      shadowGrad.addColorStop(1, 'rgba(0,0,0,0.22)');
      ctx.fillStyle = shadowGrad;
      ctx.fillRect(0, 0, W, H);

      // Vertical spine typography
      ctx.save();
      ctx.translate(W / 2, H / 2);
      ctx.rotate(Math.PI / 2);
      ctx.fillStyle = item.ink;
      ctx.textAlign = 'center';
      ctx.font = '600 34px system-ui, sans-serif';
      ctx.fillText(item.title, 0, 10);
      ctx.restore();

      const texture = new THREE.CanvasTexture(canvas);
      texture.anisotropy = 8;
      return texture;
    }

    /** Procedural page edge texture with subtle paper signatures */
    function generatePageBlockTexture() {
      const W = 512;
      const H = 256;
      const canvas = document.createElement('canvas');
      canvas.width = W;
      canvas.height = H;
      const ctx = canvas.getContext('2d');

      ctx.fillStyle = '#f5f0e6';
      ctx.fillRect(0, 0, W, H);

      // Individual page lines
      ctx.strokeStyle = 'rgba(70, 60, 45, 0.12)';
      ctx.lineWidth = 1;
      for (let x = 0; x < W; x += 3) {
        ctx.beginPath();
        ctx.moveTo(x + (Math.random() - 0.5), 0);
        ctx.lineTo(x + (Math.random() - 0.5), H);
        ctx.stroke();
      }

      // Deeper signature grouping lines
      ctx.strokeStyle = 'rgba(50, 40, 30, 0.25)';
      ctx.lineWidth = 1.5;
      for (let x = 0; x < W; x += 24) {
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, H);
        ctx.stroke();
      }

      const texture = new THREE.CanvasTexture(canvas);
      texture.wrapS = THREE.RepeatWrapping;
      texture.wrapT = THREE.RepeatWrapping;
      return texture;
    }

    const BOOK_W = 14.6;

    function createBookMesh(catalogItem) {
      const bookGroup = new THREE.Group();

      const frontCoverTex = generateCoverTexture(catalogItem);
      const spineTex = generateSpineTexture(catalogItem);

      // Book cover materials
      const clothMat = new THREE.MeshStandardMaterial({
        color: new THREE.Color(catalogItem.caseColor),
        roughness: 0.82,
        metalness: 0.04
      });

      const frontMat = new THREE.MeshStandardMaterial({
        map: frontCoverTex,
        roughness: 0.78,
        metalness: 0.02
      });

      const spineMat = new THREE.MeshStandardMaterial({
        map: spineTex,
        roughness: 0.8,
        metalness: 0.02
      });

      const paperMat = new THREE.MeshStandardMaterial({
        map: pageBlockTex,
        color: 0xf5f0e6,
        roughness: 0.95,
        metalness: 0.0
      });

      // 1. Textblock (Pages inside)
      const blockGeo = new THREE.BoxGeometry(BOOK_W - 0.5, BOOK_H - 0.8, BOOK_T - BOARD_T * 2);
      const blockMat = [
        paperMat, // right (fore-edge)
        clothMat, // left (spine back)
        paperMat, // top
        paperMat, // bottom
        clothMat, // front
        clothMat  // back
      ];
      const textblock = new THREE.Mesh(blockGeo, blockMat);
      textblock.position.set(0.15, 0, 0);
      textblock.castShadow = true;
      textblock.receiveShadow = true;
      bookGroup.add(textblock);

      // 2. Front Board
      const boardGeo = new THREE.BoxGeometry(BOOK_W, BOOK_H, BOARD_T);
      const frontBoardMaterials = [
        clothMat, clothMat, clothMat, clothMat,
        frontMat, // Front Face
        clothMat  // Inside Flyleaf
      ];
      const frontBoard = new THREE.Mesh(boardGeo, frontBoardMaterials);
      frontBoard.position.set(0, 0, (BOOK_T - BOARD_T) / 2);
      frontBoard.castShadow = true;
      frontBoard.receiveShadow = true;
      bookGroup.add(frontBoard);

      // 3. Back Board
      const backBoardMaterials = [
        clothMat, clothMat, clothMat, clothMat,
        clothMat, // Inside Flyleaf
        clothMat  // Back Face
      ];
      const backBoard = new THREE.Mesh(boardGeo, backBoardMaterials);
      backBoard.position.set(0, 0, -(BOOK_T - BOARD_T) / 2);
      backBoard.castShadow = true;
      backBoard.receiveShadow = true;
      bookGroup.add(backBoard);

      // 4. Spine Cover
      const spineGeo = new THREE.BoxGeometry(BOARD_T, BOOK_H, BOOK_T);
      const spineMaterials = [
        clothMat, spineMat, clothMat, clothMat, clothMat, clothMat
      ];
      const spine = new THREE.Mesh(spineGeo, spineMaterials);
      spine.position.set(-BOOK_W / 2, 0, 0);
      spine.castShadow = true;
      spine.receiveShadow = true;
      bookGroup.add(spine);

      bookGroup.userData = { catalog: catalogItem };
      return bookGroup;
    }

    const RING_RADIUS = 60;
    const RING_TILT = THREE.MathUtils.degToRad(9);

    const ringGroup = new THREE.Group();
    ringGroup.rotation.x = RING_TILT;
    scene.add(ringGroup);

    const bookSlots = [];

    for (let i = 0; i < N_SLOTS; i++) {
      const catalogItem = CATALOG[i % CATALOG.length];
      const bookMesh = createBookMesh(catalogItem);
      
      const angle = (i / N_SLOTS) * Math.PI * 2;
      const x = Math.sin(angle) * RING_RADIUS;
      const z = Math.cos(angle) * RING_RADIUS;

      const slotHost = new THREE.Group();
      slotHost.position.set(x, 0, z);
      // Tangential alignment to ring radius
      slotHost.rotation.y = angle + Math.PI / 2;

      slotHost.add(bookMesh);
      ringGroup.add(slotHost);

      bookSlots.push({
        index: i,
        catalog: catalogItem,
        host: slotHost,
        mesh: bookMesh,
        angle: angle
      });
    }

    /* ==========================================================================
       INTERACTION, ROTATION & UX PHYSICS
       ========================================================================== */
    let ringAngle = 0;
    let targetAngle = 0;
    let isDragging = false;
    let lastPointerX = 0;
    let pointerVelocity = 0;
    let currentBookIndex = 0;
    let inspectMode = false;

    const panel = document.getElementById('panel');
    const panelTitle = document.getElementById('panelTitle');
    const panelPromise = document.getElementById('panelPromise');
    const inspectBtn = document.getElementById('inspectBtn');
    const returnBtn = document.getElementById('returnBtn');

    function updatePresentationCard(index) {
      const item = CATALOG[((index % CATALOG.length) + CATALOG.length) % CATALOG.length];
      panelTitle.textContent = item.title;
      panelPromise.textContent = item.promise;
      panel.classList.add('active');
    }

    function commitNearestSlot() {
      const step = (Math.PI * 2) / N_SLOTS;
      const nearestSlot = Math.round(targetAngle / step);
      targetAngle = nearestSlot * step;
      currentBookIndex = ((-nearestSlot % N_SLOTS) + N_SLOTS) % N_SLOTS;
      updatePresentationCard(currentBookIndex);
    }

    // Pointer / Touch Events
    window.addEventListener('pointerdown', (e) => {
      if (e.target.closest('#panel') || e.target.closest('header') || e.target.closest('.orbit-controls')) return;
      isDragging = true;
      lastPointerX = e.clientX;
      pointerVelocity = 0;
    });

    window.addEventListener('pointermove', (e) => {
      if (!isDragging) return;
      const deltaX = e.clientX - lastPointerX;
      lastPointerX = e.clientX;
      const rotDelta = (deltaX / window.innerWidth) * 4.5;
      targetAngle += rotDelta;
      pointerVelocity = rotDelta;
    });

    window.addEventListener('pointerup', () => {
      if (!isDragging) return;
      isDragging = false;
      // Add subtle momentum fling
      targetAngle += pointerVelocity * 3;
      commitNearestSlot();
    });

    // Mouse Wheel Navigation
    let wheelTimeout;
    window.addEventListener('wheel', (e) => {
      if (inspectMode) return;
      const step = (Math.PI * 2) / N_SLOTS;
      targetAngle += Math.sign(e.deltaY) * step;
      clearTimeout(wheelTimeout);
      wheelTimeout = setTimeout(commitNearestSlot, 120);
    }, { passive: true });

    // UI Buttons
    document.getElementById('prevBtn').addEventListener('click', () => {
      const step = (Math.PI * 2) / N_SLOTS;
      targetAngle += step;
      commitNearestSlot();
    });

    document.getElementById('nextBtn').addEventListener('click', () => {
      const step = (Math.PI * 2) / N_SLOTS;
      targetAngle -= step;
      commitNearestSlot();
    });

    // Inspect / Hold Book Mode
    let inspectedMesh = null;
    inspectBtn.addEventListener('click', () => {
      inspectMode = true;
      inspectBtn.style.display = 'none';
      returnBtn.style.display = 'block';
    });

    returnBtn.addEventListener('click', () => {
      inspectMode = false;
      inspectBtn.style.display = 'block';
      returnBtn.style.display = 'none';
    });

    // Theme Switcher
    const themeBtn = document.getElementById('themeBtn');
    let isDark = false;
    themeBtn.addEventListener('click', () => {
      isDark = !isDark;
      document.documentElement.setAttribute('data-theme', isDark ? 'dark' : 'light');
      themeBtn.textContent = isDark ? 'Switch to Light' : 'Switch to Dark';
      scene.background.set(isDark ? 0x0e0d0c : 0xffffff);
      hemiLight.groundColor.set(isDark ? 0x111111 : 0x555555);
    });

    // Raycasting for Click-To-Front
    const raycaster = new THREE.Raycaster();
    const mouse = new THREE.Vector2();

    window.addEventListener('click', (e) => {
      if (isDragging || Math.abs(pointerVelocity) > 0.005) return;
      if (e.target.closest('#panel') || e.target.closest('header') || e.target.closest('.orbit-controls')) return;

      mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
      mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
      raycaster.setFromCamera(mouse, camera);

      const intersects = raycaster.intersectObjects(ringGroup.children, true);
      if (intersects.length > 0) {
        let hit = intersects[0].object;
        while (hit.parent && hit.parent !== ringGroup) {
          hit = hit.parent;
        }
        const hitSlot = bookSlots.find(s => s.host === hit);
        if (hitSlot) {
          const step = (Math.PI * 2) / N_SLOTS;
          targetAngle = -hitSlot.angle;
          commitNearestSlot();
        }
      }
    });

    // Initialize Presentation
    commitNearestSlot();

    /* ==========================================================================
       ANIMATION & RENDER LOOP
       ========================================================================== */
    const clock = new THREE.Clock();

    function animate() {
      requestAnimationFrame(animate);

      // Smooth damped spring interpolation for ring rotation
      ringAngle += (targetAngle - ringAngle) * 0.1;
      ringGroup.rotation.y = ringAngle;

      // Dynamic hover/lift and presentation for the front-most book
      const step = (Math.PI * 2) / N_SLOTS;
      for (const slot of bookSlots) {
        const effectiveAngle = ((slot.angle + ringAngle) % (Math.PI * 2) + Math.PI * 2) % (Math.PI * 2);
        const distFromFront = Math.min(effectiveAngle, Math.PI * 2 - effectiveAngle);

        if (distFromFront < step * 0.8) {
          // Front presented book: lift out and turn toward camera
          const t = 1 - (distFromFront / (step * 0.8));
          slot.mesh.position.y = THREE.MathUtils.lerp(0, 4.5, t);
          slot.mesh.position.z = THREE.MathUtils.lerp(0, 10, t);
          slot.mesh.rotation.y = THREE.MathUtils.lerp(0, -Math.PI / 2, t);
        } else {
          slot.mesh.position.y = THREE.MathUtils.lerp(slot.mesh.position.y, 0, 0.1);
          slot.mesh.position.z = THREE.MathUtils.lerp(slot.mesh.position.z, 0, 0.1);
          slot.mesh.rotation.y = THREE.MathUtils.lerp(slot.mesh.rotation.y, 0, 0.1);
        }
      }

      // Inspect camera transition
      if (inspectMode) {
        camera.position.lerp(new THREE.Vector3(0, 18, 55), 0.07);
        camera.lookAt(0, 4, 0);
      } else {
        camera.position.lerp(new THREE.Vector3(0, 38, 145), 0.07);
        camera.lookAt(0, 0, 0);
      }

      renderer.render(scene, camera);
    }

    animate();

    // Window Resize
    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });
  </script>
